Buyer beware in the quarterback portal

High transfer ratings didn’t guarantee better play for college football’s biggest 2025 offseason moves.

library(tidyverse)

portal <- read_csv("data/portal_qbs_2025.csv")
pass24 <- read_csv("data/passing_2024.csv") |>
  mutate(player = gsub("*", "", Player, fixed = TRUE)) |>
  select(player, Rate) |>
  rename(rate_24 = Rate)
pass25 <- read_csv("data/passing_2025.csv") |>
  mutate(player = gsub("*", "", Player, fixed = TRUE)) |>
  select(player, Rate, Yds, TD, Int) |>
  rename(rate_25 = Rate, yds_25 = Yds, td_25 = TD, int_25 = Int)
qbs <- portal |>
  inner_join(pass24) |>
  inner_join(pass25)

qbs <- qbs |>
  mutate(diff = rate_25 - rate_24)

ggplot() +
  geom_segment(
    data = qbs,
    aes(
      y = reorder(player, diff), yend = player,
      x = rate_24, xend = rate_25
    ),
    color = "lightgrey",
    linewidth = 1.5
  ) +
  geom_point(
    data = qbs,
    aes(y = player, x = rate_24, color = "2024"),
    size = 3.5
  ) +
  geom_point(
    data = qbs,
    aes(y = player, x = rate_25, color = "2025"),
    size = 3.5
  ) +
  scale_color_manual(values = c("red", "darkgreen")) +
  labs(
    title = "Big-name transfer quarterbacks didn't always pay off",
    subtitle = "If 2025 (green) is to the right of 2024 (red), the QB improved. If it's to the left, he regressed.",
    x = "Passer Efficiency Rating",
    y = "",
    caption = "Source: Sports Reference & 247Sports | By: Caleb Krueger"
  ) +
  theme_minimal() +
  theme(
    legend.position = "none",
    plot.title = element_text(size = 16, face = "bold"),
    plot.subtitle = element_text(size = 10),
    axis.title = element_text(size = 8),
    panel.grid.minor = element_blank(),
    plot.title.position = "plot", 
  ggsave("image.png")
  )
